% Gaussian integral through polar coordinates
% Assumptions: Integrands are nonnegative; Tonelli permits the product integral.
% Define the positive integral.
\[
I=\int_{-\infty}^{\infty}e^{-x^2}\,dx
\]
% Square it as an integral over the plane.
\[
I^2=\int_{-\infty}^{\infty}\int_{-\infty}^{\infty}e^{-x^2-y^2}\,dx\,dy
\]
% Change to polar coordinates.
\[
x=r\cos\theta,\quad y=r\sin\theta
\]
% Include the Jacobian.
\[
dx\,dy=r\,dr\,d\theta
\]
% The radial and angular factors separate.
\[
I^2=\int_0^{2\pi}\int_0^{\infty}e^{-r^2}r\,dr\,d\theta
\]
% Evaluate the radial integral.
\[
I^2=2\pi\left[-\frac{1}{2}e^{-r^2}\right]_0^{\infty}=\pi
\]
% Take the positive root.
\[
I=\sqrt{\pi}
\]
